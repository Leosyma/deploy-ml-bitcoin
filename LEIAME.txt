# Projeto 2 - Construção e Deploy de API - Machine Learning Para Prever o Preço do Bitcoin
# Instruções para executar o projeto

# Abra o terminal ou prompt de comando, navegue até a pasta com os arquivos e execute o comando abaixo para criar um ambiente virtual:

conda create --name bitcoin-api python=3.12

# Ative o ambiente:

conda activate bitcoin-api (ou: source activate bitcoin-api)

# Instale o pip e as dependências:

conda install pip
pip install -r requirements.txt 

# Abra o terminal ou prompt de comando, navegue até a pasta onde estão os arquivos e inicie a API com o comando abaixo:

python app.py

# Abra outro terminal ou prompt de comando, navegue até a pasta onde estão os arquivos, ative o ambiente virtual com as dependências (conda activate bitcoin-api) e execute a app cliente com o comando abaixo:

python cliente.py

# Use os comandos abaixo para desativar o ambiente virtual e remover o ambiente (opcional):

conda deactivate
conda remove --name bitcoin-api --all
